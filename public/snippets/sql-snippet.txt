-- Multi-Tenant Ledger & Real-Time Sync Schema
-- Architecture: PostgreSQL 16+ with Row-Level Security & Partial Indices

CREATE TABLE tenant_commercial_contracts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id VARCHAR(64) NOT NULL,
    external_unit_code VARCHAR(32) NOT NULL,
    monthly_rate NUMERIC(12, 2) NOT NULL CHECK (monthly_rate >= 0),
    cfdi_usage_code VARCHAR(8) NOT NULL DEFAULT 'G03',
    status VARCHAR(24) NOT NULL DEFAULT 'PENDING_PROVISION',
    metadata JSONB NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Compound index for sub-millisecond multi-tenant lookups
CREATE INDEX idx_tenant_status_created 
ON tenant_commercial_contracts (tenant_id, status, created_at DESC);
